<script>
    // --- データ管理 ---
    const trivias = [
        "CPUは再起動するとメモリ内のゴミが掃除されて元気になります。",
        "キーボードのFとJに突起があるのは、手元を見ずに打つための目印だよ。",
        "1GB（ギガバイト）は、実は約10億バイトという膨大なデータの集まりです。",
        "ノートPCのバッテリーは、0%や100%の状態を長く続けるより、50%前後が一番長持ちします。"
    ];

    // --- 背景泡アニメーション ---
    const canvas = document.getElementById('bg-canvas');
    const ctx = canvas.getContext('2d');
    let particles = [];

    function initCanvas() {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
    }

    class Bubble {
        constructor() {
            this.reset();
        }
        reset() {
            this.x = Math.random() * canvas.width;
            this.y = canvas.height + Math.random() * 100;
            this.size = Math.random() * 15 + 2;
            this.speed = Math.random() * 1 + 0.5;
            this.opacity = Math.random() * 0.5;
        }
        update() {
            this.y -= this.speed;
            if (this.y < -50) this.reset();
        }
        draw() {
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            ctx.fillStyle = `rgba(77, 166, 255, ${this.opacity})`;
            ctx.fill();
        }
    }

    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        if (localStorage.getItem('mode') !== 'perf') {
            particles.forEach(p => { p.update(); p.draw(); });
        }
        requestAnimationFrame(animate);
    }

    // --- システム基本機能 ---
    let screenHistory = ['lock-screen'];

    function showScreen(id) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        document.getElementById(id).classList.add('active');
        screenHistory.push(id);
    }

    function goBack() {
        if (localStorage.getItem('mode') !== 'perf') {
            createRipple();
        }
        screenHistory.pop();
        const prev = screenHistory[screenHistory.length - 1];
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        document.getElementById(prev).classList.add('active');
    }

    function createRipple() {
        const ripple = document.createElement('div');
        ripple.className = 'ripple';
        ripple.style.left = (window.innerWidth/2 - 50) + 'px';
        ripple.style.top = (window.innerHeight/2 - 50) + 'px';
        ripple.style.width = '100px';
        ripple.style.height = '100px';
        document.body.appendChild(ripple);
        setTimeout(() => ripple.remove(), 600);
    }

    function checkPassword() {
        const input = document.getElementById('pass-input').value;
        if (input === '0132') {
            showScreen('home-screen');
            document.getElementById('trivia-text').innerText = "本日の豆知識: " + trivias[Math.floor(Math.random() * trivias.length)];
        } else {
            document.getElementById('error-msg').innerText = "パスワード間違っているよ！";
        }
    }

    // --- テスト機能 ---
    let currentQuiz = [];
    let quizIndex = 0;
    let score = 0;
    let wrongAnswers = [];

    function startTest(difficulty) {
        // 本来は難易度ごとにDBから取得
        // デモ用サンプル問題
        currentQuiz = [
            { q: "メモリ8GBと16GB、どちらが一度にたくさんのアプリを開けますか？", a: ["16GB", "8GB", "変わらない"], ok: 0, exp: "まな板（メモリ）が広いほど、たくさんの料理（アプリ）を同時に作れるんだよ！" },
            { q: "Intel CPUの『Pコア』とは何の略？", a: ["Performance", "Power", "Private"], ok: 0, exp: "性能重視のベテランシェフ（Performance）のことだね。" },
            { q: "JEITA規格とは主に何に関する測定基準？", a: ["バッテリー駆動時間", "画面の明るさ", "キーボードの重さ"], ok: 0, exp: "業界共通の物差し（基準）で、正確に比較するためのものだよ。" }
            // ここに360問分のデータが入ります
        ];
        quizIndex = 0; score = 0; wrongAnswers = [];
        showScreen('test-run-screen');
        nextQuestion();
    }

    function nextQuestion() {
        if (quizIndex >= 3) { // デモなので3問で終了
            finishTest();
            return;
        }
        document.getElementById('test-progress').innerText = `${quizIndex + 1} / 20`;
        const q = currentQuiz[quizIndex];
        document.getElementById('question-text').innerText = q.q;
        const choicesArea = document.getElementById('choices-area');
        choicesArea.innerHTML = '';
        q.a.forEach((choice, i) => {
            const btn = document.createElement('div');
            btn.className = 'choice-btn';
            btn.innerText = choice;
            btn.onclick = () => selectChoice(i);
            choicesArea.appendChild(btn);
        });
    }

    function selectChoice(idx) {
        if (idx === currentQuiz[quizIndex].ok) {
            score++;
        } else {
            wrongAnswers.push({ q: currentQuiz[quizIndex].q, correct: currentQuiz[quizIndex].a[currentQuiz[quizIndex].ok], exp: currentQuiz[quizIndex].exp });
        }
        quizIndex++;
        nextQuestion();
    }

    function finishTest() {
        showScreen('result-screen');
        document.getElementById('result-score').innerText = `正解数: ${score} / 20`;
        const details = document.getElementById('result-details');
        details.innerHTML = '<h3>間違えたところの解説:</h3>';
        wrongAnswers.forEach(item => {
            details.innerHTML += `<div style="margin-bottom:15px; border-left:4px solid #ff4d4d; padding-left:10px;">
                <strong>問: ${item.q}</strong><br>
                正解: ${item.correct}<br>
                解説: ${localStorage.getItem('guide') === 'off' ? '（専門的）内部構造に基づいた正解です。' : item.exp}
            </div>`;
        });
    }

    // --- 設定保存 ---
    function updateSettings() {
        localStorage.setItem('bgColor', document.getElementById('bg-select').value);
        localStorage.setItem('mode', document.getElementById('mode-select').value);
        localStorage.setItem('guide', document.getElementById('guide-select').value);
        applySettings();
    }

    function applySettings() {
        const bg = localStorage.getItem('bgColor') || '#1a1a1a';
        document.body.style.backgroundColor = bg;
        document.querySelectorAll('.screen').forEach(s => s.style.backgroundColor = bg);
        
        const guide = localStorage.getItem('guide') || 'on';
        document.querySelectorAll('.guide-text').forEach(t => {
            t.style.display = (guide === 'on') ? 'block' : 'none';
        });
    }

    // 初期化
    window.onload = () => {
        initCanvas();
        for (let i = 0; i < 30; i++) particles.push(new Bubble());
        animate();
        applySettings();
        // セレクトボックスの値を復元
        if(localStorage.getItem('bgColor')) document.getElementById('bg-select').value = localStorage.getItem('bgColor');
        if(localStorage.getItem('mode')) document.getElementById('mode-select').value = localStorage.getItem('mode');
        if(localStorage.getItem('guide')) document.getElementById('guide-select').value = localStorage.getItem('guide');
    };
    window.onresize = initCanvas;

</script>
